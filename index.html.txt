  redeemedCodes: []

  { name: 'هوجين (محارب الشتاء)', shards: 100, unlocked: true, level: 1, img: 'https://images.unsplash.com/photo-1578632767115-351597cf2477?auto=format&fit=crop&w=200&q=80' },

];

var availableAvatars = [
  'https://images.unsplash.com/photo-1514539079130-25950c84af65?auto=format&fit=crop&w=400&q=80',
  'https://images.unsplash.com/photo-1578632767115-351597cf2477?auto=format&fit=crop&w=200&q=80',

];

function addVipExp(amount) {
  playerData.vipExp += amount;
  var newVip = Math.floor(playerData.vipExp / 500000) + 1;
  if (newVip > 11) newVip = 11;
  playerData.vipLevel = newVip;
}

function calculatePower() {
  var total = 0;
  for(var i=0; i<heroes.length; i++) {
    if(heroes[i].unlocked) {
      total += (heroes[i].level * 1500) + 2500;
    }
  }
  return total;
}

function updateUI() {
  document.getElementById('goldVal').innerText = playerData.gold.toLocaleString();
  document.getElementById('gemVal').innerText = playerData.gems.toLocaleString();
  document.getElementById('meatVal').innerText = playerData.meat.toLocaleString();
  document.getElementById('powerVal').innerText = calculatePower().toLocaleString();
  document.getElementById('stageNum').innerText = playerData.stage;
  document.getElementById('profId').innerText = playerData.id;
  document.getElementById('headerAvatarBox').innerHTML = '<img src="' + playerData.selectedAvatar + '"><div class="vip-badge" id="headerVipTag">VIP ' + playerData.vipLevel + '</div>';
  document.getElementById('profVipLevel').innerText = 'VIP ' + playerData.vipLevel;
}

function openModal(id) {
  closeModals();
  document.getElementById(id).classList.add('show');
  if(id === 'mHeroes') renderHeroes();
  if(id === 'mShop') renderShop();
  if(id === 'mRank') renderRank();
  if(id === 'mProfile') renderProfileOptions();
}

function closeModals() {
  var modals = document.querySelectorAll('.page-modal');
  modals.forEach(function(m){ m.classList.remove('show'); });
  setNav('nHome');
}

function setNav(id) {
  document.querySelectorAll('.nav-btn').forEach(function(b){ b.classList.remove('active'); });
  document.getElementById(id).classList.add('active');
}

function copyId() {
  navigator.clipboard.writeText(playerData.id);
  alert('تم نسخ معرف اللاعب (ID) بنجاح!');
}

function redeemCode() {
  var codeInput = document.getElementById('promoCodeInput').value.trim().toUpperCase();
  if (codeInput === 'SAM305') {
    if (playerData.redeemedCodes.includes('SAM305')) {
      alert('لقد قمت باستخدام هذا الكود مسبقاً!');
      return;
    }
    playerData.redeemedCodes.push('SAM305');
    playerData.gems += 90000000;
    playerData.gold += 90000000;
    playerData.meat += 90000000;
    addVipExp(2000000);
    updateUI();
    alert('🎉 مبروك! تم استرداد الكود SAM305 بنجاح والحصول على 90,000,000 من الجواهر والذهب واللحم وارتفاع مستوى الـ VIP!');
    document.getElementById('promoCodeInput').value = '';
    closeModals();
  } else {
    alert('عذراً، هذا الكود غير صحيح أو منتهي الصلاحية.');
  }
}

function renderProfileOptions() {
  var avGrid = document.getElementById('avatarOptionsGrid');
  var avHtml = '';
  for(var i=0; i<availableAvatars.length; i++) {
    var sel = (playerData.selectedAvatar === availableAvatars[i]) ? 'selected' : '';
    avHtml += '<img src="' + availableAvatars[i] + '" class="opt-thumb ' + sel + '" onclick="changeAvatar(\'' + availableAvatars[i] + '\')">';
  }
  avGrid.innerHTML = avHtml;
}

function changeAvatar(url) {
  playerData.selectedAvatar = url;
  renderProfileOptions();
  updateUI();
}

function renderHeroes() {
  var cont = document.getElementById('heroesContainer');
  var html = '';
  for(var i=0; i<heroes.length; i++) {
    var h = heroes[i];
    html += '<div class="card-item">' +
      '<div style="display:flex; align-items:center; gap:8px;">' +
        '<img src="' + h.img + '" style="width:36px; height:36px; border-radius:50%; object-fit:cover; border:1px solid #f59e0b;">' +
        '<div>' +
          '<div style="font-weight:bold; color:#f59e0b; font-size:11px;">' + h.name + '</div>' +
          '<div style="color:#38bdf8; font-size:10px;">المستوى: ' + h.level + ' | القطع: ' + h.shards + '/100</div>' +
        '</div>' +
      '</div>';
    if(h.unlocked) {
      html += '<button onclick="upgradeHero(' + i + ')" style="background:#f59e0b; color:#000; border:none; padding:5px 8px; border-radius:6px; font-weight:bold; cursor:pointer; font-size:10px;">ترقية (🍖 5,000)</button>';
    } else {
      var canUnlock = h.shards >= 100;
      html += '<button onclick="unlockHero(' + i + ')" style="background:' + (canUnlock ? '#10b981' : '#374151') + '; color:#fff; border:none; padding:5px 8px; border-radius:6px; font-weight:bold; cursor:' + (canUnlock ? 'pointer' : 'not-allowed') + '; font-size:10px;">فتح البطل</button>';
    }
    html += '</div>';
  }
  cont.innerHTML = html;
}

function upgradeHero(idx) {
  if(playerData.meat >= 5000) {
    playerData.meat -= 5000;
    heroes[idx].level += 1;
    updateUI();
    renderHeroes();
    alert('تم ترقية البطل بنجاح وزيادة القوة!');
  } else {
    alert('عذراً، لا توجد قطع لحم كافية لديك!');
  }
}

function unlockHero(idx) {
  if(heroes[idx].shards >= 100) {
    heroes[idx].unlocked = true;
    heroes[idx].level = 1;
    updateUI();
    renderHeroes();
    alert('مبارك! تم فتح البطل الجديد بنجاح وانضم لجيشك.');
  } else {
    alert('تحتاج إلى جمع 100 قطعة من المتجر لفتح هذا البطل.');
  }
}

function renderShop() {
  var cont = document.getElementById('shopContainer');
  var html = '';
  for(var i=0; i<heroes.length; i++) {
    var h = heroes[i];
    html += '<div class="card-item">' +
      '<div style="display:flex; align-items:center; gap:8px;">' +
        '<img src="' + h.img + '" style="width:36px; height:36px; border-radius:50%; object-fit:cover; border:1px solid #d97706;">' +
        '<div>' +
          '<div style="font-weight:bold; color:#f59e0b; font-size:11px;">قطعة بطل: ' + h.name + '</div>' +
          '<div style="color:#fbbf24; font-size:10px;">السعر: 🪙 80,000 ذهب</div>' +
        '</div>' +
      '</div>' +
      '<button onclick="buyShard(' + i + ')" style="background:#d97706; color:#000; border:none; padding:5px 10px; border-radius:6px; font-weight:bold; cursor:pointer; font-size:10px;">شراء قطعة</button>' +
    '</div>';
  }
  cont.innerHTML = html;
}

function buyShard(idx) {
  if(playerData.gold >= 80000) {
    playerData.gold -= 80000;
    addVipExp(80000);
    heroes[idx].shards += 1;
    updateUI();
    renderShop();
    alert('تم شراء قطعة البطل بنجاح! رصيدك الحالي من القطع: ' + heroes[idx].shards);
  } else {
    alert('عذراً، ليس لديك ذهب كافٍ (تحتاج إلى 80,000 ذهب).');
  }
}

function renderRank() {
  var cont = document.getElementById('rankContainer');
  var myP = calculatePower();
  var serverPlayers = [
    { name: 'MARENA [ETR]', power: 445000, avatar: 'https://images.unsplash.com/photo-1507003211169-0a1dd7228f2d?auto=format&fit=crop&w=200&q=80' },
    { name: 'Hades7777 [PY]', power: 378000, avatar: 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?auto=format&fit=crop&w=200&q=80' },
    { name: 'راغنار [SWAT]', power: myP, avatar: playerData.selectedAvatar, isMe: true },
    { name: 'Doruu [VGN]', power: 215000, avatar: 'https://images.unsplash.com/photo-1578632767115-351597cf2477?auto=format&fit=crop&w=200&q=80' }
  ];
  serverPlayers.sort(function(a, b) { return b.power - a.power; });

  var html = '';
  for(var i=0; i<serverPlayers.length; i++) {
    var p = serverPlayers[i];
    var bg = p.isMe ? '#0c4a6e' : '#1f2937';
    var border = p.isMe ? '1px solid #38bdf8' : '1px solid #374151';
    var rankNum = i + 1;
    var rankColor = rankNum === 1 ? '#fbbf24' : (rankNum === 2 ? '#9ca3af' : '#fff');

    html += '<div class="card-item" style="background:' + bg + '; border:' + border + ';">' +
      '<div style="display:flex; align-items:center; gap:8px;">' +
        '<span style="font-weight:bold; font-size:13px; color:' + rankColor + '; width:18px; text-align:center;">#' + rankNum + '</span>' +
        '<img src="' + p.avatar + '" style="width:32px; height:32px; border-radius:50%; object-fit:cover; border:1px solid #f59e0b;">' +
        '<div>' +
          '<div style="font-weight:bold; color:#fff; font-size:11px;">' + p.name + (p.isMe ? ' (أنت)' : '') + '</div>' +
          '<div style="color:#38bdf8; font-size:10px;">⚔️ القوة: ' + p.power.toLocaleString() + '</div>' +
        '</div>' +
      '</div>' +
      '<span style="font-size:10px; color:#f59e0b; font-weight:bold;">الخادم #1</span>' +
    '</div>';
  }
  cont.innerHTML = html;
}

function fightStage() {
  var enemyPower = playerData.stage <= 10 ? 5000 : (5000 + (playerData.stage - 10) * 1200);
  var myPower = calculatePower();

  if(myPower >= enemyPower) {
    playerData.gold += 40000;
    playerData.gems += 20;
    playerData.meat += 15000;
    if(playerData.stage < 300) playerData.stage += 1;
    updateUI();
    alert('انتصار ملحمي! 🎉 لقد هزمت أعداء المرحلة وحصلت على:\n🪙 40,000 ذهب\n💎 20 جوهرة\n🍖 15,000 لحم');
    closeModals();
  } else {
    alert('تم صد هجومك! قوتك الحالية (' + myPower.toLocaleString() + ') أقل من قوة أعداء المرحلة (' + enemyPower.toLocaleString() + '). قم بتطوير أبطالك أولاً!');
  }
}

function claimMail() {
  playerData.gold += 200000;

  playerData.meat += 50000;

  document.getElementById('mailBadge').style.display = 'none';
  alert('تم إضافة الهدايا إلى رصيدك بنجاح! 🎁');
}
